**Step 1: Write `eslint-rules/no-raw-process-env.js`** with this EXACT content:

```javascript
/**
 * Custom ESLint rule: no-raw-process-env
 *
 * Enforces D-11 from Phase 1 (Scaffold + Brand Token Parity):
 *   Raw reads of `process.env` must go through `lib/env.ts`. Every other
 *   module that needs an env value imports `{ env } from "@/lib/env"` —
 *   the typed Zod-parsed object — so that the boot-time crash invariant
 *   (D-08, D-10) cannot be silently bypassed.
 *
 * Detection surface:
 *   - `MemberExpression` visitor flags any node where `object.name === 'process'`
 *     AND `property.name === 'env'`. This catches both `process.env.X` and
 *     destructuring like `const { X } = process.env`. Computed access
 *     (`process['env']`) is also caught when both children are Identifier nodes.
 *
 * Allowlist (rule returns no visitors → fires nothing):
 *   - `lib/env.ts` — the one sanctioned process.env reader
 *   - `eslint-rules/**` — this file and its tests
 *   - `*.test.ts`, `*.test.tsx`, `*.test.js` — test fixtures may stub env
 */

'use strict'

const ALLOWLIST_FRAGMENTS = [
  '/lib/env.ts',
  '/eslint-rules/',
]
const TEST_FILE_SUFFIXES = ['.test.ts', '.test.tsx', '.test.js']

function normalizeFilename(name) {
  return (name ?? '').replace(/\\/g, '/')
}

function isAllowlisted(filename) {
  if (TEST_FILE_SUFFIXES.some((s) => filename.endsWith(s))) return true
  return ALLOWLIST_FRAGMENTS.some((f) => filename.includes(f))
}

/** @type {import('eslint').Rule.RuleModule} */
module.exports = {
  meta: {
    type: 'problem',
    docs: {
      description:
        'Raw process.env reads must go through lib/env.ts (D-11). Import { env } from "@/lib/env" instead.',
      recommended: false,
    },
    messages: {
      raw:
        'Do not read process.env directly. Import { env } from "@/lib/env" — the Zod-validated env object — instead. (D-11)',
    },
    schema: [],
  },

  create(context) {
    const rawFilename = context.filename ?? context.getFilename()
    const filename = normalizeFilename(rawFilename)
    if (isAllowlisted(filename)) return {}

    return {
      // Match `process.env` in any context: `process.env.X`, `const { X } = process.env`,
      // `Object.keys(process.env)`, `process['env']`, etc.
      MemberExpression(node) {
        const isProcessEnv =
          node.object &&
          node.object.type === 'Identifier' &&
          node.object.name === 'process' &&
          ((node.property.type === 'Identifier' && node.property.name === 'env') ||
            (node.property.type === 'Literal' && node.property.value === 'env'))
        if (isProcessEnv) {
          context.report({ node, messageId: 'raw' })
        }
      },
    }
  },
}
```

**Step 3: Write `.gitleaks.toml`** with this EXACT content (per D-14 + INFRA-08 success criterion):

```toml
# Quibly Landing — gitleaks custom ruleset
# Extends gitleaks' built-in default rules with patterns for the
# specific secrets this repo will handle (Resend + Upstash, Phase 4).
#
# Reference: D-14 in 01-CONTEXT.md.
# Pre-commit invocation: gitleaks protect --staged --redact -c .gitleaks.toml

[extend]
useDefault = true

# Resend API keys: re_ + 20+ base62 chars
# Verified empirically — Resend keys begin with "re_" prefix.
[[rules]]
id = "resend-api-key"
description = "Resend API key"
regex = '''re_[A-Za-z0-9]{20,}'''
keywords = ["re_"]

# Upstash REST API tokens: AYxA prefix + 30+ base64-url chars
[[rules]]
id = "upstash-rest-token"
description = "Upstash REST API token"
regex = '''AYxA[A-Za-z0-9_-]{30,}'''
keywords = ["AYxA"]

# Upstash REST URL host (catches accidental .env.local commits)
[[rules]]
id = "upstash-rest-url"
description = "Upstash REST URL"
regex = '''https://[a-z0-9-]+\.upstash\.io'''
keywords = ["upstash.io"]

# Allowlist: documentation files and the .env.example template
# (the placeholder values in .env.example match our regex by length but
#  are deliberately deterministic placeholders — never real secrets).
[allowlist]
description = "Documentation and example files — placeholders only"
paths = [
  '''(.+/)?\.env\.example$''',
  '''(.+/)?\.gitleaks\.toml$''',
  '''(.+/)?CLAUDE\.md$''',
  '''(.+/)?README\.md$''',
  '''(.+/)?\.planning/.*\.md$''',
]
# Also allowlist the literal placeholder strings used in .env.example
# so an inline `cat .env.example | git apply` style edit cannot trick the
# regex into thinking the placeholder is a real secret.
regexes = [
  '''re_xxxxxxxxxxxxxxxxxxxxx''',
  '''AYxAxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxx''',
  '''your-instance\.upstash\.io''',
]
```

**Critical:** the `[allowlist]` block is what prevents `.env.example` from blocking its OWN commit. The placeholder `re_xxxxxxxxxxxxxxxxxxxxx` (21 x's) matches the `re_[A-Za-z0-9]{20,}` regex by length, so without the allowlist it would block. Both path-based AND regex-based allowlisting are used as defense-in-depth.
