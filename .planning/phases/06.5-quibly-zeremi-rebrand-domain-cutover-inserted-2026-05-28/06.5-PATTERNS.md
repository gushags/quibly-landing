5. Inline the two `<g transform="...">` groups from `zeremi-mark.svg` (three `<path>` elements total: Z letter + two dots)

```tsx
export function QuibsAvatar({ size, inverted = false, className }: QuibsAvatarProps) {

    <div
      className={cn(
        'flex items-center justify-center',
        container,
        palette,
        className,
      )}
    >
      <QuibsIcon width={icon.width} height={icon.height} />
    </div>
  )
}
```
